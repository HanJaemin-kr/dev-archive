# JavaScript 문법 정리

## 입출력

```js
const readline = require('readline');
const rl = readline.createInterface({ input: process.stdin });
const lines = [];
rl.on('line', (line) => lines.push(line.trim()));
rl.on('close', () => {
  // 여기서 풀기
});
```

프로그래머스는 `readline` 없이 함수 인자로 바로 들어옴.

## 반복문

### 한눈에 비교

| 방법          | 반환값               | break | 인덱스 | 용도                            |
| ----------- | ----------------- | ----- | --- | ----------------------------- |
| `for`       | —                 | ✅     | ✅   | 인덱스 필요, 조기 탈출 필요              |
| `for...of`  | —                 | ✅     | ❌   | 값만 순회, 조기 탈출 필요               |
| `for...in`  | —                 | ✅     | key | 객체 키 순회 (배열엔 쓰지 말 것)          |
| `while`     | —                 | ✅     | 직접  | 조건 기반 반복, 탈출 조건 복잡할 때         |
| `forEach`   | `undefined`       | ❌     | ✅   | 부작용(출력, 상태 변경), 중단 불필요        |
| `map`       | 새 배열              | ❌     | ✅   | 1:1 변환, 원본 불변                 |
| `filter`    | 새 배열              | ❌     | ✅   | 조건으로 걸러내기                     |
| `reduce`    | 누적값               | ❌     | ✅   | 합산·빈도수·객체 변환 등                |
| `find`      | 요소 or `undefined` | —     | ✅   | 첫 번째 일치 요소                    |
| `findIndex` | 인덱스 or `-1`       | —     | ✅   | 첫 번째 일치 인덱스                   |
| `some`      | `boolean`         | ✅(내부) | ✅   | 하나라도 조건 만족 / early exit 트릭    |
| `every`     | `boolean`         | ✅(내부) | ✅   | 모두 조건 만족                      |
| `flatMap`   | 새 배열              | ❌     | ✅   | map + flat(1), 요소를 여러 개로 펼칠 때 |

### 기본 루프

```js
// for — 인덱스 필요할 때, break/continue 필요할 때
for (let i = 0; i < n; i++) { }

// for...of — 값만 필요할 때 (배열, Set, Map, 문자열 모두 가능)
for (const x of arr) { }
for (const ch of str) { }
for (const [k, v] of map) { }
for (const x of set) { }

// for...in — 객체 키 순회 (배열엔 쓰지 말 것 — 프로토타입 키까지 나옴)
for (const key in obj) { }

// while
while (condition) { }

// 역방향
for (let i = arr.length - 1; i >= 0; i--) { }
```

### 배열 고차 함수

```js
// forEach — 반환값 없음, break 불가
arr.forEach((v, i) => { });

// map — 새 배열 반환, 원본 불변
const doubled = arr.map(v => v * 2);
const matrix = arr.map((v, i) => [i, v]);

// filter — 조건 만족하는 요소만
const evens = arr.filter(v => v % 2 === 0);

// reduce — 누적 연산
const sum = arr.reduce((acc, v) => acc + v, 0);
const max = arr.reduce((acc, v) => Math.max(acc, v), -Infinity);
// 빈도수 객체 만들기
const freq = arr.reduce((acc, v) => {
  acc[v] = (acc[v] ?? 0) + 1;
  return acc;
}, {});

// find / findIndex — 첫 번째 일치, 없으면 undefined / -1
const found = arr.find(v => v > 3);
const idx = arr.findIndex(v => v > 3);

// some / every — 하나라도 / 모두
arr.some(v => v > 0);
arr.every(v => v > 0);

// flatMap — map + flat(1) 합본
[[1,2],[3,4]].flatMap(x => x);  // [1, 2, 3, 4]
arr.flatMap(v => [v, v * 2]);   // 각 요소를 2개로 펼치기
```

### 언제 뭘 쓰나

| 상황 | 선택 |
|------|------|
| 인덱스 접근 or break 필요 | `for` |
| 값만 순회, break 필요 | `for...of` |
| 부작용(출력, 상태 변경) | `forEach` |
| 변환된 새 배열 필요 | `map` |
| 조건으로 걸러내기 | `filter` |
| 하나의 값으로 합치기 | `reduce` |
| 객체 키 순회 | `for...in` |

### 주의사항

```js
// forEach는 break/return으로 중단 불가 → for...of 또는 some 활용
arr.forEach(v => {
  if (v === 3) return; // 이 iteration만 skip, 전체 중단 아님
});

// 중단이 필요하면
for (const v of arr) {
  if (v === 3) break; // 완전 중단 가능
}

// 또는 some으로 early exit 흉내
arr.some(v => {
  if (v === 3) return true; // true 반환 시 순회 중단
  doSomething(v);
});

// map은 원본을 변경하지 않음
const original = [1, 2, 3];
const doubled = original.map(v => v * 2); // original은 그대로
```

## 스택 / 큐

```js
// 스택 — 배열로 충분
const stack = [];
stack.push(x);
stack.pop();

// 큐 — shift()는 O(n)이므로 큰 입력엔 직접 구현
class Queue {
  constructor() { this.q = {}; this.head = 0; this.tail = 0; }
  enqueue(v) { this.q[this.tail++] = v; }
  dequeue() { if (this.isEmpty()) return; const v = this.q[this.head]; delete this.q[this.head++]; return v; }
  isEmpty() { return this.head === this.tail; }
  size() { return this.tail - this.head; }
}
```

## 재귀 / DFS / BFS

```js
// DFS (재귀)
function dfs(node, visited) {
  visited.add(node);
  for (const next of graph[node]) {
    if (!visited.has(next)) dfs(next, visited);
  }
}

// BFS
function bfs(start) {
  const visited = new Set([start]);
  const queue = [start];
  while (queue.length) {
    const cur = queue.shift();
    for (const next of graph[cur]) {
      if (!visited.has(next)) {
        visited.add(next);
        queue.push(next);
      }
    }
  }
}
```

## 이진 탐색

```js
// 정렬된 배열에서 target의 삽입 위치 (lower bound)
function lowerBound(arr, target) {
  let lo = 0, hi = arr.length;
  while (lo < hi) {
    const mid = (lo + hi) >> 1;
    if (arr[mid] < target) lo = mid + 1;
    else hi = mid;
  }
  return lo;
}
```

// 순열 (재귀)
function permute(arr, cur = []) {
  if (cur.length === arr.length) { result.push([...cur]); return; }
  for (let i = 0; i < arr.length; i++) {
    if (used[i]) continue;
    used[i] = true; cur.push(arr[i]);
    permute(arr, cur);
    used[i] = false; cur.pop();
  }
}

// 조합 (재귀)
function combine(arr, k, start = 0, cur = []) {
  if (cur.length === k) { result.push([...cur]); return; }
  for (let i = start; i < arr.length; i++) {
    cur.push(arr[i]);
    combine(arr, k, i + 1, cur);
    cur.pop();
  }
}

## 클래스

### 기본 선언

```js
class Person {
  // 필드 (ES2022+) — 생성자 없이도 초기값 지정 가능
  species = 'human';
  #secret = 'hidden';  // private 필드 (#)

  // 생성자
  constructor(name, age) {
    this.name = name;
    this.age = age;
  }

  // 인스턴스 메서드
  greet() {
    return `Hi, I'm ${this.name}`;
  }

  // getter / setter
  get info() { return `${this.name}(${this.age})`; }
  set info(v) { [this.name, this.age] = v.split(','); }

  // 정적 메서드 — 인스턴스 없이 클래스에서 직접 호출
  static create(name) { return new Person(name, 0); }

  // 정적 필드
  static count = 0;
}

const p = new Person('Tom', 20);
p.greet();              // 'Hi, I'm Tom'
p.info;                 // 'Tom(20)' (getter)
Person.create('Amy');   // 정적 메서드
Person.count;           // 정적 필드
```

### 상속 (extends / super)

```js
class Animal {
  constructor(name) { this.name = name; }
  speak() { return `${this.name} makes a sound`; }
}

class Dog extends Animal {
  constructor(name, breed) {
    super(name);            // 부모 생성자 호출 — 반드시 this 사용 전에
    this.breed = breed;
  }
  speak() {
    return super.speak() + ' (bark)';  // 부모 메서드 호출
  }
}

const d = new Dog('Rex', 'Husky');
d.speak();   // 'Rex makes a sound (bark)'
d instanceof Dog;     // true
d instanceof Animal;  // true
```

### 정리 표

| 키워드 | 용도 |
|--------|------|
| `class` | 클래스 선언 |
| `constructor` | 인스턴스 초기화 (한 클래스에 하나만) |
| `static` | 인스턴스가 아닌 클래스에 속하는 멤버 |
| `#field` | private 필드 (외부 접근 불가, 문법 에러) |
| `extends` | 상속 |
| `super(...)` | 부모 생성자 호출 |
| `super.method()` | 부모 메서드 호출 |
| `get` / `set` | 접근자 — 메서드를 속성처럼 사용 |

### 주의사항

```js
// 1. 클래스 선언은 호이스팅되지만 TDZ에 들어감 → 선언 전 사용 불가
new Foo();           // ReferenceError
class Foo {}

// 2. 클래스 내부는 자동 strict mode
// 3. 메서드는 prototype에 정의 — 모든 인스턴스가 공유
class A { m() {} }
const a1 = new A(), a2 = new A();
a1.m === a2.m;       // true (prototype에 단 하나)

// 4. 화살표 함수 필드 — this 바인딩 고정 (이벤트 핸들러에 유용)
class Btn {
  text = 'ok';
  onClick = () => console.log(this.text);  // this가 항상 인스턴스
}

// 5. 정적 메서드 안의 this는 클래스 자체
class C {
  static x = 1;
  static log() { console.log(this.x); }  // this === C
}
```

## 메모


- Q: `sort()` 기본 동작의 문제점은?
  A: 유니코드 코드포인트 기준 정렬 → 숫자 배열에서 `[10, 2, 1]` → `[1, 10, 2]`. 반드시 비교 함수 전달.

- Q: `==` vs `===` 차이?
  A: `==`는 타입 강제 변환 후 비교, `===`는 타입까지 일치해야 함. 코딩 테스트에선 항상 `===` 사용.

- Q: `const` 배열/객체가 수정 가능한 이유?
  A: `const`는 바인딩(참조)을 고정하는 것이지 값을 고정하는 것이 아님.
